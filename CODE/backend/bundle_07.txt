#pragma once
#include <string>
#include <chrono>
#include <nlohmann/json.hpp>

namespace models {

struct Corporation {
    int id = 0;
    int user_id = 0;
    std::string name;
    double budget = 1000000.0;
    int reputation = 50;
    int current_season = 1;
    int current_week = 1;
    bool esports_active = true;
    bool racing_active = false;
    bool tactical_active = false;
    bool innovation_active = false;
    std::chrono::system_clock::time_point created_at;
    std::chrono::system_clock::time_point updated_at;

    nlohmann::json toJson() const {
        return nlohmann::json{
            {"id", id},
            {"user_id", user_id},
            {"name", name},
            {"budget", budget},
            {"reputation", reputation},
            {"current_season", current_season},
            {"current_week", current_week},
            {"divisions", {
                {"esports_active", esports_active},
                {"racing_active", racing_active},
                {"tactical_active", tactical_active},
                {"innovation_active", innovation_active}
            }},
            {"created_at", std::chrono::duration_cast<std::chrono::seconds>(created_at.time_since_epoch()).count()},
            {"updated_at", std::chrono::duration_cast<std::chrono::seconds>(updated_at.time_since_epoch()).count()}
        };
    }
};

// Ajout : fonction to_json pour nlohmann::json
inline void to_json(nlohmann::json& j, const models::Corporation& corp) {
    j = corp.toJson();
}

} // namespace models

#pragma once
#include <string>
#include <chrono>
#include <nlohmann/json.hpp>

namespace models {

struct User {
    int id = 0;
    std::string username;
    std::string email;
    std::string password_hash;
    bool is_active = true;
    std::chrono::system_clock::time_point created_at;
    std::chrono::system_clock::time_point updated_at;

    // Convert to JSON (without password hash for security)
    nlohmann::json toJson() const {
        return nlohmann::json{
            {"id", id},
            {"username", username},
            {"email", email},
            {"is_active", is_active},
            {"created_at", std::chrono::duration_cast<std::chrono::seconds>(created_at.time_since_epoch()).count()},
            {"updated_at", std::chrono::duration_cast<std::chrono::seconds>(updated_at.time_since_epoch()).count()}
        };
    }
};

} // namespace models

#pragma once
#include <memory>
#include <future>
#include <string>
#include "repositories/UserRepository.h"
#include "dto/auth/LoginRequest.h"
#include "dto/auth/RegisterRequest.h"
#include "dto/auth/AuthResponse.h"
#include "utils/PasswordUtils.h"
#include "utils/JwtUtils.h"
#include "utils/Logger.h"

namespace services {

class AuthService {
private:
    std::shared_ptr<repositories::UserRepository> userRepo_;

public:
    explicit AuthService(std::shared_ptr<repositories::UserRepository> userRepo) 
        : userRepo_(std::move(userRepo)) {}

    // Register new user
    std::future<dto::auth::AuthResponse> registerUserAsync(const dto::auth::RegisterRequest& request) {
        return std::async(std::launch::async, [this, request]() -> dto::auth::AuthResponse {
            try {
                // Validate request
                auto validationErrors = request.validate();
                if (!validationErrors.empty()) {
                    utils::Logger::warn("Registration validation failed for user: " + request.username);
                    return dto::auth::AuthResponse::createError("Validation failed: " + validationErrors[0]);
                }

                // Check if username already exists
                auto usernameExists = userRepo_->usernameExistsAsync(request.username).get();
                if (usernameExists) {
                    utils::Logger::warn("Registration failed - username already exists: " + request.username);
                    return dto::auth::AuthResponse::createError("Username already exists");
                }

                // Check if email already exists
                auto emailExists = userRepo_->emailExistsAsync(request.email).get();
                if (emailExists) {
                    utils::Logger::warn("Registration failed - email already exists: " + request.email);
                    return dto::auth::AuthResponse::createError("Email already exists");
                }

                // Hash password
                std::string passwordHash = utils::PasswordUtils::hashPassword(request.password);

                // Create user
                auto userOpt = userRepo_->createUserAsync(request.username, request.email, passwordHash).get();
                if (!userOpt.has_value()) {
                    utils::Logger::error("Failed to create user in database: " + request.username);
                    return dto::auth::AuthResponse::createError("Failed to create user account");
                }

                // Generate JWT token
                std::string token = utils::JwtUtils::generateToken(userOpt.value());
                if (token.empty()) {
                    utils::Logger::error("Failed to generate JWT token for user: " + request.username);
                    return dto::auth::AuthResponse::createError("Failed to generate authentication token");
                }

                utils::Logger::info("User registered successfully: " + request.username);
                return dto::auth::AuthResponse::createSuccess(token, userOpt.value());

            } catch (const std::exception& e) {
                utils::Logger::error("Exception in registerUser: " + std::string(e.what()));
                return dto::auth::AuthResponse::createError("Internal server error");
            }
        });
    }

    // Login user
    std::future<dto::auth::AuthResponse> loginUserAsync(const dto::auth::LoginRequest& request) {
        return std::async(std::launch::async, [this, request]() -> dto::auth::AuthResponse {
            try {
                // Validate request
                auto validationErrors = request.validate();
                if (!validationErrors.empty()) {
                    utils::Logger::warn("Login validation failed for user: " + request.username);
                    return dto::auth::AuthResponse::createError("Invalid request: " + validationErrors[0]);
                }

                // Find user by username
                auto userOpt = userRepo_->findUserByUsernameAsync(request.username).get();
                if (!userOpt.has_value()) {
                    utils::Logger::warn("Login failed - user not found: " + request.username);
                    return dto::auth::AuthResponse::createError("Invalid credentials");
                }

                auto user = userOpt.value();

                // Verify password
                if (!utils::PasswordUtils::verifyPassword(request.password, user.password_hash)) {
                    utils::Logger::warn("Login failed - invalid password for user: " + request.username);
                    return dto::auth::AuthResponse::createError("Invalid credentials");
                }

                // Generate JWT token
                std::string token = utils::JwtUtils::generateToken(user);
                if (token.empty()) {
                    utils::Logger::error("Failed to generate JWT token for user: " + request.username);
                    return dto::auth::AuthResponse::createError("Failed to generate authentication token");
                }

                utils::Logger::info("User logged in successfully: " + request.username);
                return dto::auth::AuthResponse::createSuccess(token, user);

            } catch (const std::exception& e) {
                utils::Logger::error("Exception in loginUser: " + std::string(e.what()));
                return dto::auth::AuthResponse::createError("Internal server error");
            }
        });
    }

    // Verify JWT token and get user
    std::future<std::optional<models::User>> verifyTokenAndGetUserAsync(const std::string& token) {
        return std::async(std::launch::async, [this, token]() -> std::optional<models::User> {
            try {
                auto payload = utils::JwtUtils::verifyToken(token);
                if (!payload.has_value()) {
                    return std::nullopt;
                }

                // Get fresh user data from database
                return userRepo_->findUserByIdAsync(payload.value().user_id).get();

            } catch (const std::exception& e) {
                utils::Logger::error("Exception in verifyTokenAndGetUser: " + std::string(e.what()));
                return std::nullopt;
            }
        });
    }
};

} // namespace services
