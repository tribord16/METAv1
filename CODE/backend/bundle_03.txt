#pragma once
#include <memory>
#include <functional>
#include <optional>
#include <future>
#include "models/User.h"
#include "utils/DatabaseConnection.h"
#include "utils/Logger.h"

namespace repositories {

class UserRepository {
private:
    std::shared_ptr<utils::DatabaseConnection> dbPool_;

public:
    explicit UserRepository(std::shared_ptr<utils::DatabaseConnection> dbPool) 
        : dbPool_(std::move(dbPool)) {}

    // Create new user asynchronously
    std::future<std::optional<models::User>> createUserAsync(const std::string& username, 
                                                             const std::string& email, 
                                                             const std::string& passwordHash) {
        return std::async(std::launch::async, [this, username, email, passwordHash]() -> std::optional<models::User> {
            try {
                utils::ConnectionGuard conn(*dbPool_);
                
                // Check if username or email already exists
                std::unique_ptr<sql::PreparedStatement> checkStmt(
                    conn->prepareStatement("SELECT COUNT(*) as count FROM users WHERE username = ? OR email = ?"));
                checkStmt->setString(1, username);
                checkStmt->setString(2, email);
                
                std::unique_ptr<sql::ResultSet> checkRes(checkStmt->executeQuery());
                if (checkRes->next() && checkRes->getInt("count") > 0) {
                    utils::Logger::warn("Attempt to create user with existing username/email: " + username);
                    return std::nullopt;
                }
                
                // Insert new user
                std::unique_ptr<sql::PreparedStatement> stmt(
                    conn->prepareStatement(
                        "INSERT INTO users (username, email, password_hash, is_active, created_at, updated_at) "
                        "VALUES (?, ?, ?, TRUE, NOW(), NOW())"
                    ));
                
                stmt->setString(1, username);
                stmt->setString(2, email);
                stmt->setString(3, passwordHash);
                
                int affectedRows = stmt->executeUpdate();
                if (affectedRows == 0) {
                    utils::Logger::error("Failed to insert user: " + username);
                    return std::nullopt;
                }
                
                // Get the created user
                return findUserByUsernameSync(username);
                
            } catch (sql::SQLException& e) {
                utils::Logger::error("Database error in createUser: " + std::string(e.what()));
                return std::nullopt;
            }
        });
    }

    // Find user by username asynchronously
    std::future<std::optional<models::User>> findUserByUsernameAsync(const std::string& username) {
        return std::async(std::launch::async, [this, username]() -> std::optional<models::User> {
            return findUserByUsernameSync(username);
        });
    }

    // Find user by email asynchronously
    std::future<std::optional<models::User>> findUserByEmailAsync(const std::string& email) {
        return std::async(std::launch::async, [this, email]() -> std::optional<models::User> {
            try {
                utils::ConnectionGuard conn(*dbPool_);
                
                std::unique_ptr<sql::PreparedStatement> stmt(
                    conn->prepareStatement(
                        "SELECT id, username, email, password_hash, is_active, "
                        "UNIX_TIMESTAMP(created_at) as created_at, "
                        "UNIX_TIMESTAMP(updated_at) as updated_at "
                        "FROM users WHERE email = ? AND is_active = TRUE"
                    ));
                
                stmt->setString(1, email);
                std::unique_ptr<sql::ResultSet> res(stmt->executeQuery());
                
                if (res->next()) {
                    return resultSetToUser(*res);
                }
                
                return std::nullopt;
                
            } catch (sql::SQLException& e) {
                utils::Logger::error("Database error in findUserByEmail: " + std::string(e.what()));
                return std::nullopt;
            }
        });
    }

    // Find user by ID asynchronously
    std::future<std::optional<models::User>> findUserByIdAsync(int userId) {
        return std::async(std::launch::async, [this, userId]() -> std::optional<models::User> {
            try {
                utils::ConnectionGuard conn(*dbPool_);
                
                std::unique_ptr<sql::PreparedStatement> stmt(
                    conn->prepareStatement(
                        "SELECT id, username, email, password_hash, is_active, "
                        "UNIX_TIMESTAMP(created_at) as created_at, "
                        "UNIX_TIMESTAMP(updated_at) as updated_at "
                        "FROM users WHERE id = ? AND is_active = TRUE"
                    ));
                
                stmt->setInt(1, userId);
                std::unique_ptr<sql::ResultSet> res(stmt->executeQuery());
                
                if (res->next()) {
                    return resultSetToUser(*res);
                }
                
                return std::nullopt;
                
            } catch (sql::SQLException& e) {
                utils::Logger::error("Database error in findUserById: " + std::string(e.what()));
                return std::nullopt;
            }
        });
    }

    // Check if username exists
    std::future<bool> usernameExistsAsync(const std::string& username) {
        return std::async(std::launch::async, [this, username]() -> bool {
            try {
                utils::ConnectionGuard conn(*dbPool_);
                
                std::unique_ptr<sql::PreparedStatement> stmt(
                    conn->prepareStatement("SELECT COUNT(*) as count FROM users WHERE username = ?"));
                stmt->setString(1, username);
                
                std::unique_ptr<sql::ResultSet> res(stmt->executeQuery());
                if (res->next()) {
                    return res->getInt("count") > 0;
                }
                
                return false;
                
            } catch (sql::SQLException& e) {
                utils::Logger::error("Database error in usernameExists: " + std::string(e.what()));
                return false;
            }
        });
    }

    // Check if email exists
    std::future<bool> emailExistsAsync(const std::string& email) {
        return std::async(std::launch::async, [this, email]() -> bool {
            try {
                utils::ConnectionGuard conn(*dbPool_);
                
                std::unique_ptr<sql::PreparedStatement> stmt(
                    conn->prepareStatement("SELECT COUNT(*) as count FROM users WHERE email = ?"));
                stmt->setString(1, email);
                
                std::unique_ptr<sql::ResultSet> res(stmt->executeQuery());
                if (res->next()) {
                    return res->getInt("count") > 0;
                }
                
                return false;
                
            } catch (sql::SQLException& e) {
                utils::Logger::error("Database error in emailExists: " + std::string(e.what()));
                return false;
            }
        });
    }

private:
    // Synchronous helper for finding user by username
    std::optional<models::User> findUserByUsernameSync(const std::string& username) {
        try {
            utils::ConnectionGuard conn(*dbPool_);
            
            std::unique_ptr<sql::PreparedStatement> stmt(
                conn->prepareStatement(
                    "SELECT id, username, email, password_hash, is_active, "
                    "UNIX_TIMESTAMP(created_at) as created_at, "
                    "UNIX_TIMESTAMP(updated_at) as updated_at "
                    "FROM users WHERE username = ? AND is_active = TRUE"
                ));
            
            stmt->setString(1, username);
            std::unique_ptr<sql::ResultSet> res(stmt->executeQuery());
            
            if (res->next()) {
                return resultSetToUser(*res);
            }
            
            return std::nullopt;
            
        } catch (sql::SQLException& e) {
            utils::Logger::error("Database error in findUserByUsername: " + std::string(e.what()));
            return std::nullopt;
        }
    }

    // Convert SQL ResultSet to User model
    models::User resultSetToUser(sql::ResultSet& res) {
        models::User user;
        user.id = res.getInt("id");
        user.username = res.getString("username");
        user.email = res.getString("email");
        user.password_hash = res.getString("password_hash");
        user.is_active = res.getBoolean("is_active");
        
        // Convert UNIX timestamp to time_point
        auto created_timestamp = res.getInt64("created_at");
        auto updated_timestamp = res.getInt64("updated_at");
        
        user.created_at = std::chrono::system_clock::from_time_t(created_timestamp);
        user.updated_at = std::chrono::system_clock::from_time_t(updated_timestamp);
        
        return user;
    }
};

} // namespace repositories
